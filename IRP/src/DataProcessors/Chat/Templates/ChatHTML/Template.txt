  <script>
    // === messages array (example JSON) ===
    const messages = #MessageArray#;

    // Helper function for date formatting (dd.mm.yyyy)
    function formatDate(date) {
      return date.toLocaleDateString("ru-RU");
    }

    // Helper function for time formatting (HH:MM)
    function formatTime(date) {
      return date.toLocaleTimeString("ru-RU", { hour: '2-digit', minute: '2-digit' });
    }

    // Rendering messages
    function renderMessages() {
      const chatContainer = document.getElementById("chat-container");
      let previousDate = null;

      messages.forEach(msg => {
        const dateObj = new Date(msg.Period);
        const msgDate = formatDate(dateObj);

        // Show a date divider if the date changes
        if (msgDate !== previousDate) {
          const divider = document.createElement('div');
          divider.className = 'date-divider';
          divider.textContent = msgDate;
          chatContainer.appendChild(divider);
          previousDate = msgDate;
        }

        // If the message is system (ManualComment = false)
        if (!msg.ManualComment) {
          const systemBubble = document.createElement('div');
          systemBubble.classList.add('message-bubble', 'system-message', 'clearfix');
          
          // System message includes user and datetime
          systemBubble.innerHTML = `
            <span class="system-info">
              System message from ${msg.User} at ${msgDate} ${formatTime(dateObj)}
            </span>
            <span class="system-text">
              ${msg.Message}
            </span>
          `;
          chatContainer.appendChild(systemBubble);
        } 
        else {
          // Normal user message
          const bubble = document.createElement("div");
          bubble.classList.add("message-bubble", "clearfix");
          
          // Decide if it's outgoing or incoming
          if (msg.CurrentUser) {
            bubble.classList.add("outgoing");
          } else {
            bubble.classList.add("incoming");
          }

          // Add user label and time
          bubble.innerHTML = `
            <div class="message-header">
              <span class="user-label">${msg.User}</span>
              <span class="time-label">${formatTime(dateObj)}</span>
            </div>
            <div class="message-content">${msg.Message.replace(/\n/g, '<br>')}</div>
          `;
          chatContainer.appendChild(bubble);
        }
      });

      // Auto-scroll to the bottom after all messages are rendered
      chatContainer.scrollTop = chatContainer.scrollHeight;
    }

    // When DOM is loaded, render messages
    window.addEventListener("DOMContentLoaded", renderMessages);
    chatContainer.scrollTop = chatContainer.scrollHeight;
  </script>
