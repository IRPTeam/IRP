  <script>
    const messages = #MessageArray#;
    function formatDate(date) {
      return date.toLocaleDateString("en-EN");
    }
    function formatTime(date) {
      return date.toLocaleTimeString("en-EN", { hour: '2-digit', minute: '2-digit' });
    }
    function renderNotifyUsers(notifies) {
      if (!notifies || !notifies.length) return "";
      let result = `<div class="system-meta"><table class="notify-table">`;
      notifies.forEach(n => {
        const allFalse = !n.SendEmail && !n.WaitForOpenRef && !n.RefOpened;
        if (allFalse) return;
        let iconsHtml = "";
        if (n.SendEmail) {
          if (n.Sended) {
            iconsHtml += `<span class="icon" title="Email sent">&#10004;</span>`;
          } else {
            iconsHtml += `<span class="icon" title="Email not sent">&#10008;</span>`;
          }
        }
        if (n.WaitForOpenRef) {
          if (n.RefOpened) {
            iconsHtml += `<span class="icon" title="Document viewed">&#9673;</span>`;
          } else {
            iconsHtml += `<span class="icon" title="Waiting to be viewed">&#9203;</span>`;
          }
        }
        if (iconsHtml) {
          result += `
            <tr>
              <td class="notify-user-name">${n.User}</td>
              <td class="notify-icons">${iconsHtml}</td>
            </tr>`;
        }
      });
      result += `</table></div>`;
      return result;
    }
    function renderMessages() {
      const chatContainer = document.getElementById("chat-container");
      let previousDate = null;
      messages.forEach(msg => {
        const dateObj = new Date(msg.Period);
        const msgDate = formatDate(dateObj);
        if (msgDate !== previousDate) {
          const divider = document.createElement("div");
          divider.className = "date-divider";
          divider.textContent = msgDate;
          chatContainer.appendChild(divider);
          previousDate = msgDate;
        }
        if (!msg.ManualComment) {
          const systemBubble = document.createElement("div");
          systemBubble.classList.add("message-bubble", "system-message", "clearfix");
          const notifyBlock = renderNotifyUsers(msg.Notify);
          systemBubble.innerHTML = `
            <div class="system-info">System message from ${msg.User} at ${msgDate} ${formatTime(dateObj)}</div>
            <div class="system-text">${msg.Message}</div>
            ${notifyBlock}
          `;
          chatContainer.appendChild(systemBubble);
        } else {
          const bubble = document.createElement("div");
          bubble.classList.add("message-bubble", "clearfix");
          if (msg.CurrentUser) bubble.classList.add("outgoing");
          else bubble.classList.add("incoming");
          bubble.innerHTML = `
            <div class="message-header">
              <span class="user-label">${msg.User}</span>
              <span class="time-label">${formatTime(dateObj)}</span>
            </div>
            <div class="message-content">${msg.Message}</div>
          `;
          chatContainer.appendChild(bubble);
        }
      });
      chatContainer.scrollTop = chatContainer.scrollHeight;
    }
    window.addEventListener("DOMContentLoaded", renderMessages);
  </script>
